// Pipeline stage names for progress bar
const STAGE_NAMES = {
1: 'Extract', 2: 'Research', 3: 'Project Plan', 4: 'Proposal',
5: 'AI Report', 6: 'LLM Fill', 7: 'Render', 8: 'Polish', 9: 'PDF'
};
const TOTAL_STAGES = 9;

function updateProgress(currentStage, status = 'running') {
const container = document.getElementById('progress-container');
const fill = document.getElementById('progress-bar-fill');
const stageName = document.getElementById('progress-stage-name');
const percent = document.getElementById('progress-percent');

container.classList.add('active');
fill.classList.remove('complete', 'error');
const pct = Math.round((currentStage / TOTAL_STAGES) * 100);
fill.style.width = pct + '%';
if (status === 'error') {
fill.classList.add('error');
stageName.textContent = `Failed: ${STAGE_NAMES[currentStage] || 'Unknown'}`;
} else {
stageName.textContent = STAGE_NAMES[currentStage] || 'Processing...';
}
percent.textContent = pct + '%';
}

function resetProgress() {

}

function completeProgress(success = true) {

function queueExecution() {
const input = document.getElementById('input-text').value;
if (!input) return alert("Input required");
generateBtn.disabled = true;
statusLabel.innerText = "INITIALIZING...";
consoleEl.innerHTML = '<div style="color:var(--sunset-300)">> Establishing connection to Wranngle Intelligence Core...</div>';
// Show action buttons and progress bar during execution
document.getElementById('console-actions').style.display = 'flex';
resetProgress();
document.getElementById('progress-container').classList.add('active');
if (eventSource) eventSource.close();
eventSource = new EventSource('/api/stream');
eventSource.onmessage = (event) => {
const data = JSON.parse(event.data);
if (data.msg) {
const line = document.createElement('div');
line.innerHTML = ansi_up.ansi_to_html(data.msg);
consoleEl.appendChild(line);
requestAnimationFrame(() => {
consoleEl.scrollTop = consoleEl.scrollHeight;
});

// Update progress bar on stage changes
const stageMatch = data.msg.match(/STAGE (\d+)/i);
if (stageMatch) {
const stage = parseInt(stageMatch[1], 10);
updateProgress(stage);
statusLabel.innerText = `STAGE ${stage}: ${STAGE_NAMES[stage] || ''}`;
}
// Handle completion
if (data.msg.includes('COMPLETED SUCCESSFULLY')) {
statusLabel.innerText = "COMPLETE";
completeProgress(true);
generateBtn.disabled = false;
refreshHistory();
} else if (data.msg.includes('FATAL ERROR') || data.msg.includes('Pipeline failed')) {
statusLabel.innerText = "FAILED";
const currentStage = parseInt(data.msg.match(/STAGE (\d+)/)?.[1] || '1', 10);
updateProgress(currentStage, 'error');
generateBtn.disabled = false;
refreshHistory();
}
}
};
fetch('/api/generate', {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ input })
}).catch(err => {
const line = document.createElement('div');
line.style.color = "var(--violet-500)";
line.innerText = `> FATAL ERROR: ${err.message}`;
consoleEl.appendChild(line);
completeProgress(false);
generateBtn.disabled = false;
});
}

function loadFile(file) {
const textExtensions = ['.txt', '.md', '.json', '.csv', '.xml', '.html', '.log', '.rtf', '.yml', '.yaml', '.ini', '.cfg'];
const ext = '.' + file.name.split('.').pop().toLowerCase();
if (!textExtensions.includes(ext) && !file.type.startsWith('text/')) {
alert('Please drop a text-based file (.txt, .md, .json, .csv, etc.)');
return;
}
const reader = new FileReader();
reader.onload = (e) => {
document.getElementById('input-text').value = e.target.result;
document.getElementById('file-label').innerText = `✓ Loaded: ${file.name}`;
document.getElementById('file-label').style.color = "var(--sunset-500)";
};
reader.readAsText(file);
}

// Handle both milestone objects and Phase objects
const name = m.name || m.milestone_name || m.phase_name || 'Unnamed Phase';
const hours = m.hours || (m.milestones ? m.milestones.reduce((sum, ms) => sum + (ms.hours || 0), 0) : 0);
const desc = m.description || '';
return `
<div class="ai-summary-box" style="margin-bottom:8px;">
<div style="display:flex; justify-content:space-between; align-items:center; margin-bottom:4px;">
<b>${name}</b>
<span class="summary-pill">${hours}h</span>
</div>
<div style="font-size:11px; opacity:0.8;">${desc}</div>
</div>
`;
}).join('');
} else {
milestoneHtml += '<p style="font-size:12px; opacity:0.6; margin-top:8px;">No milestone data found in schema.</p>';
}
document.getElementById('tab-milestones').innerHTML = milestoneHtml;

} catch (e) {
console.error("Schema load error", e);
const errHtml = `<p style="color:var(--violet-500)">Error loading data: ${e.message}</p>`;
document.getElementById('tab-research').innerHTML = errHtml;
document.getElementById('tab-milestones').innerHTML = errHtml;
document.getElementById('json-viewer-container').innerHTML = errHtml;
}
} else {
const noJsonHtml = "<p>JSON artifact not available for this execution.</p>";
document.getElementById('tab-research').innerHTML = noJsonHtml;
document.getElementById('tab-milestones').innerHTML = noJsonHtml;
document.getElementById('json-viewer-container').innerHTML = noJsonHtml;
}
document.getElementById('drilldown-modal').classList.add('open');
}

/**
* Inject copy buttons into JSON tree nodes
* @param {HTMLElement} container - The JSON viewer container
* @param {Object} rootData - The root JSON data object
*/
function injectJsonCopyButtons(container, rootData) {
// Wait for DOM to settle
setTimeout(() => {
const rows = container.querySelectorAll('.json-formatter-row');
rows.forEach((row, index) => {
// Get the key element
const keyEl = row.querySelector(':scope > .json-formatter-key');
if (!keyEl) return;
// Skip if already has copy button next to this key
if (keyEl.nextElementSibling?.classList?.contains('json-copy-btn')) return;
// Build the path by traversing up the DOM
const path = buildJsonPath(row);
// Create inline copy icon button
const btn = document.createElement('button');
btn.className = 'json-copy-btn';
btn.innerHTML = '📋';
btn.title = `Copy: ${path.join('.')}`;
btn.onclick = (e) => {
e.stopPropagation();
const value = getValueAtPath(rootData, path);
const text = typeof value === 'object' ? JSON.stringify(value, null, 2) : String(value);
navigator.clipboard.writeText(text).then(() => {
btn.innerHTML = '✓';
btn.classList.add('copied');
setTimeout(() => {
btn.innerHTML = '📋';
btn.classList.remove('copied');
}, 1500);
});
};
// Insert right after the key element
keyEl.insertAdjacentElement('afterend', btn);
});
// Also inject buttons when nodes are expanded
container.addEventListener('click', (e) => {
if (e.target.classList.contains('json-formatter-toggler')) {
setTimeout(() => injectJsonCopyButtons(container, rootData), 50);
}
});
}, 100);
}

/**
* Build JSON path by traversing DOM hierarchy
*/
function buildJsonPath(row) {
const path = [];
let current = row;
while (current && current.id !== 'json-viewer-container') {
const keyEl = current.querySelector(':scope > .json-formatter-key');
if (keyEl) {
const key = keyEl.textContent.replace(/[":]/g, '').trim();
if (key) path.unshift(key);
}
// Move up to parent row
current = current.parentElement?.closest('.json-formatter-row');
}
return path;
}

/**
* Get value at a specific path in an object
*/
function getValueAtPath(obj, path) {
let current = obj;
for (const key of path) {
if (current === null || current === undefined) return undefined;
// Handle array indices
if (Array.isArray(current) && /^\d+$/.test(key)) {
current = current[parseInt(key, 10)];
} else {
current = current[key];
}
}
return current;
}

function copyToClipboard(text, btn) {
navigator.clipboard.writeText(text).then(() => {
const original = btn.innerText;
btn.innerText = "Copied!";
btn.style.background = "var(--color-healthy)";
btn.style.borderColor = "var(--color-healthy)";
setTimeout(() => {
btn.innerText = original;
btn.style.background = "";
btn.style.borderColor = "";
}, 2000);
}).catch(err => {
btn.innerText = "Failed";
setTimeout(() => btn.innerText = "Copy", 2000);
});
}

function copyLogs(event) {
const logs = consoleEl.innerText;
copyToClipboard(logs, event.target);
}
